% Body/sans face: Inter (variable font; weights picked with the wght axis, italics are a synthetic slant
% because Google's Inter build has no slnt axis). Set here rather than via `mainfont:` because Pandoc
% mangles the nested braces of fontspec options when they come from YAML metadata.
\setmainfont{Inter-Variable.ttf}[
  Path=assets/fonts/ttf/, BoldFont=Inter-Variable.ttf, ItalicFont=Inter-Variable.ttf, BoldItalicFont=Inter-Variable.ttf,
  UprightFeatures={RawFeature={axis={wght=400}}},
  BoldFeatures={RawFeature={axis={wght=700}}},
  ItalicFeatures={FakeSlant=0.18},
  BoldItalicFeatures={RawFeature={axis={wght=700}},FakeSlant=0.18}]
\setsansfont{Inter-Variable.ttf}[
  Path=assets/fonts/ttf/, BoldFont=Inter-Variable.ttf,
  UprightFeatures={RawFeature={axis={wght=400}}},
  BoldFeatures={RawFeature={axis={wght=700}}}]

% Display face for chapter/section titles (variable font: weight picked with the wght axis).
\newfontfamily\displayfont{InstrumentSans-Variable.ttf}[
  Path=assets/fonts/ttf/, BoldFont=InstrumentSans-Variable.ttf,
  UprightFeatures={RawFeature={axis={wght=650}}},
  BoldFeatures={RawFeature={axis={wght=700}}}]
